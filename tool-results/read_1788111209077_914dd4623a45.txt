     2→// Order notifications — email (Brevo) + SMS (Termii)
     3→// =============================================================================
     4→// Sends booking confirmations and status-change updates to customers.
     5→//
     6→// Design rules:
     7→//   - NEVER throws: a failed notification must never break an order update.
     8→//     Every public function wraps its work in try/catch and logs failures.
     9→//   - Env-gated: if BREVO_API_KEY / TERMII_API_KEY are not set, the channel
    10→//     is skipped with a console warning (so the feature is safe to deploy
    11→//     before keys are configured).
    12→//   - SMS is reserved for the statuses a customer actually needs to act on
    13→//     (picked up, out for delivery, delivered) to conserve Termii credits.
    14→//     Email is sent for every status change.
    15→// =============================================================================

    42→function turnaroundCopy(speed?: string | null): string {

    44→  if (speed === 'EXPRESS_24') return 'Express 24 — back within 24 hours of pickup'
    45→  return 'Standard — back within 3–5 days'

   187→// ----- Termii SMS -----
   188→async function sendSMS(to: string, message: string): Promise<void> {
   189→  const apiKey = process.env.TERMII_API_KEY
   190→  if (!apiKey) {
   191→    console.warn('TERMII_API_KEY not set — skipping SMS send')
   192→    return
   193→  }
   194→  const senderId = process.env.TERMII_SENDER_ID || 'Kozy'
   195→  const channel = process.env.TERMII_CHANNEL || 'generic'
   196→
   197→  const res = await fetch('https://api.ng.termii.com/api/sms/send', {
   198→    method: 'POST',
   199→    headers: { 'Content-Type': 'application/json' },
   200→    body: JSON.stringify({
   201→      to: to.replace(/\s|-/g, ''),
   202→      from: senderId,
   203→      sms: message,
   204→      type: 'plain',
   205→      channel,
   206→    }),
   207→  })
   208→  if (!res.ok) {
   209→    const err = await res.text()
   210→    throw new Error(`Termii send failed (${res.status}): ${err}`)
   211→  }
   212→}

   218→// ----- Bank-transfer order: verification underway -----
   219→// Sent the moment a bank-transfer order is confirmed. This is the email that
   220→// answers "did my payment go through?" — it says clearly that verification is
   221→// underway, another email follows the moment admin confirms, and the customer
   222→// must NOT pay again.
   223→export async function notifyTransferPendingVerification(
   224→  order: NotifiableOrder
   225→): Promise<void> {
   226→  try {
   227→    const amount = order.totalPrice ?? 0
   228→    const { subject, html } = await brandedEmail({
   229→      heading: 'We’re verifying your transfer',
   230→      intro:
   231→        'Thank you for booking with Kozy Care. Your order is in and our team is verifying your bank transfer right now — usually within minutes during business hours (Mon–Sat, 8am–6pm). You’ll get another email the moment it’s confirmed. Please don’t send the transfer again or re-book: if you completed it, we have it, and your rider is dispatched as soon as payment is verified.',
   232→      order,
   233→      extraRows: [
   234→        { label: 'Amount', value: formatNaira(amount) },
   235→        { label: 'Payment', value: 'Bank transfer — being verified' },
   236→        { label: 'Narration reference', value: `Use #${order.orderNumber}` },
   237→      ],
   238→      cta: { label: 'Check payment status', url: `${baseUrl()}/payment/pending?order=${order.orderNumber}&email=${encodeURIComponent(order.user.email)}` },
   239→      footer:
   240→        'The status page updates itself while we verify — no need to refresh or resend anything.<br>Kozy Care — Uncompromising care. Exceptional convenience.',

   248→  } catch (e) {
   249→    console.error('notifyTransferPendingVerification failed:', e)
   250→  }
   251→}

   319→// ----- Status change (called from PATCH /api/orders/[id]) -----
   320→export async function notifyOrderStatus(
   321→  order: NotifiableOrder,
   322→  newStatus: string
   323→): Promise<void> {
   324→  try {
   325→    const copy = STATUS_COPY[newStatus]
   326→    if (!copy) return
   327→
   328→    // Email — every status change
   329→    const { subject, html } = await brandedEmail({
   330→      heading: copy.title,
   331→      intro: copy.body,
   332→      order,
   333→      cta:
   334→        newStatus === 'DELIVERED'
   335→          ? { label: 'Rate your experience', url: `${baseUrl()}/review/${order.id}` }
   336→          : { label: 'Track your order', url: `${baseUrl()}/portal` },
   337→    })
   338→    await sendEmail({ to: order.user.email, subject, html })
   339→
   340→    // SMS — only the actionable statuses
   341→    if (SMS_STATUSES.has(newStatus)) {
   342→      const smsText =
   343→        newStatus === 'DELIVERED'
   344→          ? `Kozy Care: Order #${order.orderNumber} delivered. Thank you! Rate your experience: ${baseUrl()}/review/${order.id}`
   345→          : `Kozy Care: ${copy.title} — order #${order.orderNumber}. ${newStatus === 'OUT_FOR_DELIVERY' ? 'Our rider is on the way to you.' : ''}`.trim()
   346→      await sendSMS(order.user.phone, smsText)
   347→    }
   348→  } catch (e) {
   349→    console.error('notifyOrderStatus failed:', e)
   350→  }
   351→}
