   328→// =====================================================
   329→// SERVICE SPEED — turnaround tiers for retail (ITEM) orders
   330→// =====================================================
   331→// Market research (2026): express dry cleaning typically runs
   332→//   • next-day / 48h express: +25–50% over standard rates
   333→//   • same-day / 24h express: +50–100% over standard rates
   334→// Kozy's client survey puts the Lagos standard at 3–5 days, so speed is
   335→// sold as a premium rather than given away free. Surcharges sit in the
   336→// upper half of the industry band because Kozy is a premium brand that
   337→// also fronts pickup + delivery logistics on every express order.
   338→export type ServiceSpeed = 'STANDARD' | 'EXPRESS_48' | 'EXPRESS_24'
   339→

   346→  surcharge: number

   349→  enabled: boolean

   588→export function formatDateTime(iso?: string): string {
   589→  if (!iso) return '—'
   590→  const d = new Date(iso)
   591→  return d.toLocaleString('en-NG', {
   592→    day: '2-digit',
   593→    month: 'short',
   594→    year: 'numeric',
   595→    hour: '2-digit',
   596→    minute: '2-digit',
   597→  })
   598→}

   622→export function buildNotification(
   623→  trigger:
   624→    | 'BOOKING_PLACED'
   625→    | 'PICKED_UP'
   626→    | 'B2B_INVOICE_READY'
   627→    | 'PROCESSING'
   628→    | 'FINISHING'
   629→    | 'OUT_FOR_DELIVERY'
   630→    | 'DELIVERED'
   631→    | 'PAYMENT_VERIFIED',
   632→  order: Order,
   633→  user: User,
   634→  driver?: User,
   635→  extra?: { weight?: number; amount?: number; url?: string }
   636→): string {
   637→  switch (trigger) {
   638→    case 'BOOKING_PLACED':
   639→      return `Hi ${user.name.split(' ')[0]}, your pickup (#${order.orderNumber}) is booked for ${formatDate(order.pickupDate)}.`
   640→    case 'PICKED_UP':
   641→      return `Goods Received! Items collected by ${driver?.name ?? 'our rider'}. Track: ${extra?.url ?? '/portal'}`
   642→    case 'B2B_INVOICE_READY':
   643→      return `Order #${order.orderNumber} weighed ${extra?.weight}kg. Total: ₦${extra?.amount?.toLocaleString('en-NG')}. Pay here: ${extra?.url ?? '/portal'}`
   644→    case 'PROCESSING':
   645→      return `Your garments are now being washed and treated. Order #${order.orderNumber}.`
   646→    case 'FINISHING':
   647→      return `Almost done! Order #${order.orderNumber} is undergoing ironing and packaging.`
   648→    case 'OUT_FOR_DELIVERY':
   649→      return `On the way! ${driver?.name ?? 'Our rider'} is delivering your laundry. Call: ${driver?.phone ?? ''}`
   650→    case 'DELIVERED':
   651→      return `Delivered! Order #${order.orderNumber} is complete. Rate your experience: ${extra?.url ?? '/rate/' + order.orderNumber}`
   652→    case 'PAYMENT_VERIFIED':
   653→      return `Payment Confirmed! ₦${(extra?.amount ?? 0).toLocaleString('en-NG')} received for Order #${order.orderNumber}. Thank you!`
   654→  }
   655→}
