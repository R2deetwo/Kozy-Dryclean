   183→   183→// ----- Termii SMS -----
   184→   184→async function sendSMS(to: string, message: string): Promise<void> {
   185→   185→  const apiKey = process.env.TERMII_API_KEY
   186→   186→  if (!apiKey) {
   187→   187→    console.warn('TERMII_API_KEY not set — skipping SMS send')
   188→   188→    return
   189→   189→  }
   190→   190→  const senderId = process.env.TERMII_SENDER_ID || 'Kozy'
   191→   191→  const channel = process.env.TERMII_CHANNEL || 'generic'
   192→   192→
   193→   193→  const res = await fetch('https://api.ng.termii.com/api/sms/send', {
   194→   194→    method: 'POST',
   195→   195→    headers: { 'Content-Type': 'application/json' },
   196→   196→    body: JSON.stringify({
   197→   197→      to: to.replace(/\s|-/g, ''),
   198→   198→      from: senderId,
   199→   199→      sms: message,
   200→   200→      type: 'plain',
   201→   201→      channel,
   202→   202→    }),
   203→   203→  })
   204→   204→  if (!res.ok) {
   205→   205→    const err = await res.text()
   206→   206→    throw new Error(`Termii send failed (${res.status}): ${err}`)
   207→   207→  }
   208→   208→}

   326→   326→      heading: copy.title,
   327→   327→      intro: copy.body,

   329→   329→      cta:
   330→   330→        newStatus === 'DELIVERED'

   334→   334→    await sendEmail({ to: order.user.email, subject, html })
   335→   335→
   336→   336→    // SMS — only the actionable statuses
   337→   337→    if (SMS_STATUSES.has(newStatus)) {
   338→   338→      const smsText =
   339→   339→        newStatus === 'DELIVERED'
   340→   340→          ? `Kozy Care: Order #${order.orderNumber} delivered. Thank you! Rate your experience: ${baseUrl()}/review/${order.id}`
   341→   341→          : `Kozy Care: ${copy.title} — order #${order.orderNumber}. ${newStatus === 'OUT_FOR_DELIVERY' ? 'Our rider is on the way to you.' : ''}`.trim()
   342→   342→      await sendSMS(order.user.phone, smsText)
   343→   343→    }
   344→   344→  } catch (e) {
   345→   345→    console.error('notifyOrderStatus failed:', e)

   411→   411→    process.env.ADMIN_ALERTS_EMAIL || settings.contactEmail || 'kozygarmentcare@gmail.com'
   412→   412→  return {

   414→   414→      ? normalizeEmail(settings.adminAlertsEmail)

   507→   507→/** A new order was placed (authed customer or guest checkout). */
   508→   508→export async function notifyAdminNewOrder(order: NotifiableOrder): Promise<void> {
   509→   509→  try {
   510→   510→    const cfg = await adminAlertConfig()
   511→   511→    if (!cfg.newOrder) return
   512→   512→    let itemCount = '—'
   513→   513→    try {
   514→   514→      const parsed = JSON.parse((order as any).itemsManifest || '[]')
   515→   515→      if (Array.isArray(parsed) && parsed.length > 0) itemCount = `${parsed.length} item${parsed.length === 1 ? '' : 's'}`
   516→   516→    } catch {
   517→   517→      /* KG orders have no manifest */
   518→   518→    }
   519→   519→    const { subject, html } = adminEmail({
   520→   520→      badge: 'New order',
   521→   521→      heading: `New order #${order.orderNumber}`,
   522→   522→      intro:
   523→   523→        order.type === 'KG'
   524→   524→          ? 'A corporate/bulk booking came in — total is quoted after weighing at the station.'
   525→   525→          : 'A new pickup booking came in. It will appear on your Orders board immediately.',
   526→   526→      rows: [
   527→   527→        { label: 'Customer', value: order.user.name },
   528→   528→        { label: 'Phone', value: order.user.phone },
   529→   529→        { label: 'Pickup', value: `${fmtDate(order.pickupDate)} · ${order.pickupTimeSlot}` },
   530→   530→        { label: 'Address', value: order.pickupAddress },
   531→   531→        { label: 'Basket', value: order.type === 'KG' ? 'Bulk (per-kg)' : itemCount },
   532→   532→        { label: 'Total', value: order.totalPrice ? formatNaira(order.totalPrice) : 'To be weighed' },
   533→   533→        {
   534→   534→          label: 'Payment',
   535→   535→          value: (order as any).payments?.some?.((p: any) => p.status === 'PENDING') || order.status === 'PAYMENT_PENDING_VERIFICATION'
   536→   536→            ? 'Bank transfer — verify it now'
   537→   537→            : 'Bank transfer / card',
   538→   538→        },
   539→   539→      ],
   540→   540→      cta: { label: 'Open the Orders board', url: `${baseUrl()}/admin` },
   541→   541→    })
   542→   542→    await sendEmail({ to: cfg.email, subject, html })
   543→   543→  } catch (e) {
   544→   544→    console.error('notifyAdminNewOrder failed:', e)
   545→   545→  }
   546→   546→}
