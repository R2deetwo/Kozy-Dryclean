    48→
    49→export function SettingsView() {
    50→  const settings = useStore((s) => s.settings)
    51→  const updateSettings = useStore((s) => s.updateSettings)
    52→  const setGarmentPrice = useStore((s) => s.setGarmentPrice)
    53→  const queryClient = useQueryClient()
    54→
    55→  // ----- Server-managed app settings (Phase 14) -----
    56→  // Bank details, commercial terms and offers live in the AppSetting table
    57→  // so an edit reaches EVERY visitor instantly — the old flow only updated
    58→  // this browser's localStorage and customers kept seeing stale details
    59→  // (the client-reported bug). Load current values from the server.
    60→  const { data: appSettings } = useQuery({
    61→    queryKey: ['app-settings'],
    62→    queryFn: async () => {
    63→      const res = await fetch('/api/settings/app')
    64→      if (!res.ok) throw new Error('Failed to load settings')
    65→      const data = await res.json()
    66→      return data.settings as KozyAppSettings
    67→    },
    68→    staleTime: 30 * 1000,
    69→  })
    70→
    71→  const [appDraft, setAppDraft] = useState<KozyAppSettings | null>(null)
    72→  // The draft starts from whatever the server currently says (once loaded).
    73→  const app = appDraft ?? appSettings
    74→  const setApp = (patch: Partial<KozyAppSettings>) =>
    75→    setAppDraft({ ...(app as KozyAppSettings), ...patch })
    76→
    77→  const [draft, setDraft] = useState<KozySettings>(settings)
    78→  const [saving, setSaving] = useState(false)
    79→  const [saved, setSaved] = useState(false)
    80→
    81→  const handleSave = async () => {
    82→    setSaving(true)
    83→    // ----- 1. Server-side app settings (bank, contact, commercial) -----
    84→    if (appDraft && appSettings) {
    85→      const changed: Partial<KozyAppSettings> = {}
    86→      ;(Object.keys(appDraft) as (keyof KozyAppSettings)[]).forEach((k) => {
    87→        if (appDraft[k] !== appSettings[k]) {
    88→          ;(changed as Record<string, unknown>)[k] = appDraft[k]
    89→        }
    90→      })
    91→      if (Object.keys(changed).length > 0) {
    92→        try {
    93→          const res = await fetch('/api/settings/app', {
    94→            method: 'PUT',
    95→            headers: { 'Content-Type': 'application/json' },
    96→            body: JSON.stringify({ settings: changed }),
    97→          })
    98→          if (!res.ok) {
    99→            const err = await res.json().catch(() => ({}))
   100→            throw new Error(err.error || 'Server rejected the settings update')
   101→          }
   102→          await queryClient.invalidateQueries({ queryKey: ['app-settings'] })
   103→        } catch (e: any) {
   104→          toast({
   105→            title: 'Settings not saved',
   106→            description: e?.message || 'Could not reach the server — nothing was changed.',
   107→            variant: 'destructive',
   108→          })
   109→          setSaving(false)
   110→          return
   111→        }
   112→      }
   113→    }
   114→
   115→    // ----- 2. Legacy local settings (B2B per-kg pricing, guarantee note) -----
   116→    updateSettings({
   117→      pricePerKg: draft.pricePerKg,
   118→      minimumKg: draft.minimumKg,
   119→      guaranteeDiscountPercent: draft.guaranteeDiscountPercent,
   120→    })
   121→    // ----- 3. Garment prices -> PriceCatalog (server-side, live for all) -----
   122→    const changedPrices: Record<string, number> = {}
   123→    Object.entries(draft.garmentPrices).forEach(([id, price]) => {
   124→      if (settings.garmentPrices[id] !== price) {
   125→        setGarmentPrice(id, price)
   126→        changedPrices[id] = price
   127→      }
   128→    })
   129→    // …and push them to the server (PriceCatalog) so they go live on the
   130→    // storefront and match what customers are charged at checkout.
   131→    if (Object.keys(changedPrices).length > 0) {
   132→      try {
   133→        const res = await fetch('/api/settings/prices', {
   134→          method: 'PUT',
   135→          headers: { 'Content-Type': 'application/json' },
   136→          body: JSON.stringify({ garmentPrices: changedPrices }),
   137→        })
   138→        if (!res.ok) {
   139→          const err = await res.json().catch(() => ({}))
   140→          throw new Error(err.error || 'Server rejected the price update')
   141→        }
   142→      } catch (e: any) {
   143→        toast({
   144→          title: 'Prices saved locally only',
   145→          description: e?.message || 'Could not reach the server — storefront prices were not updated.',
   146→          variant: 'destructive',
   147→        })
   148→        setSaving(false)
   149→        return
   150→      }
   151→    }
   152→    setAppDraft(null)
   153→    setTimeout(() => {
   154→      setSaving(false)
   155→      setSaved(true)
   156→      toast({
   157→        title: 'Settings saved',
   158→        description: 'Changes are now live across the customer-facing app.',
   159→      })
   160→      setTimeout(() => setSaved(false), 2000)
   161→    }, 600)
   162→  }
   163→
   164→  return (
   165→    <div className="p-4 sm:p-6">
   166→      <div className="mb-6 flex items-center justify-between">
   167→        <div>
   168→          <h1 className="font-serif text-2xl font-semibold tracking-tight text-navy">
   169→            Settings
   170→          </h1>
   171→          <p className="mt-1 text-sm text-navy-300">
   172→            Manage bank account, contact details, and pricing. Changes appear instantly on the
   173→            customer-facing app.
   174→          </p>
   175→        </div>
   176→        <div className="flex gap-2">
   177→          <Button
   178→            onClick={handleSave}
   179→            disabled={saving}
   180→            className={cn(
   181→              'rounded-full',
   182→              saved
   183→                ? 'bg-emerald-600 hover:bg-emerald-700'
   184→                : 'bg-gold-gradient text-navy hover:opacity-90'
   185→            )}
   186→          >
   187→            {saving ? (
   188→              'Saving...'
   189→            ) : saved ? (
   190→              <>
   191→                <CheckCircle2 className="mr-1.5 h-4 w-4" /> Saved
   192→              </>
   193→            ) : (
   194→              <>
   195→                <Save className="mr-1.5 h-4 w-4" /> Save changes
   196→              </>
   197→            )}
   198→          </Button>
   199→        </div>
   200→      </div>
   201→
   202→      <Tabs defaultValue="bank">
   203→        <TabsList className="bg-linen-200">
   204→          <TabsTrigger value="bank" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   205→            <Banknote className="mr-1.5 h-3.5 w-3.5" /> Bank Account
   206→          </TabsTrigger>
   207→          <TabsTrigger value="contact" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   208→            <Phone className="mr-1.5 h-3.5 w-3.5" /> Contact
   209→          </TabsTrigger>
   210→          <TabsTrigger value="pricing" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   211→            <Tag className="mr-1.5 h-3.5 w-3.5" /> Pricing
   212→          </TabsTrigger>
   213→          <TabsTrigger value="terms" className="data-[state=active]:bg-navy data-[state=active]:text-white">
   214→            <Truck className="mr-1.5 h-3.5 w-3.5" /> Offers & Delivery
   215→          </TabsTrigger>

   217→            <Percent className="mr-1.5 h-3.5 w-3.5" /> Guarantee

   224→        {/* BANK ACCOUNT TAB — server-backed (Phase 14): edits are saved to

   227→          <Card className="border-navy-100 shadow-navy">

   230→                <Building2 className="h-4 w-4 text-gold-400" /> Bank Account Details

   233→                These details appear during checkout when customers choose

   236→              </p>
   237→            </CardHeader>
   238→            <CardContent className="space-y-4">
   239→              {!app ? (
   240→                <p className="py-6 text-center text-sm text-navy-300">Loading current account details…</p>
   241→              ) : (
   242→                <>
   243→                  <div>
   244→                    <Label htmlFor="bank-name" className="text-xs uppercase tracking-wide text-navy-300">
   245→                      Bank Name
   246→                    </Label>
   247→                    <Input
   248→                      id="bank-name"
   249→                      value={app.bankName}
   250→                      onChange={(e) => setApp({ bankName: e.target.value })}
   251→                      placeholder="e.g. Guaranty Trust Bank (GTB)"
   252→                      className="mt-1.5"
   253→                    />
   254→                  </div>
   255→                  <div>
   256→                    <Label htmlFor="account-name" className="text-xs uppercase tracking-wide text-navy-300">
   257→                      Account Name
   258→                    </Label>
   259→                    <Input
   260→                      id="account-name"
   261→                      value={app.accountName}
   262→                      onChange={(e) => setApp({ accountName: e.target.value })}
   263→                      placeholder="e.g. Kozy Premium Dry Cleaning Ltd"
   264→                      className="mt-1.5"
   265→                    />
   266→                  </div>
   267→                  <div>
   268→                    <Label htmlFor="account-number" className="text-xs uppercase tracking-wide text-navy-300">
   269→                      Account Number
   270→                    </Label>
   271→                    <Input
   272→                      id="account-number"
   273→                      value={app.accountNumber}
   274→                      onChange={(e) => setApp({ accountNumber: e.target.value })}
   275→                      placeholder="10-digit account number"
   276→                      className="mt-1.5 font-mono"
   277→                    />
   278→                  </div>
   279→
   280→                  {/* Preview */}
   281→                  <div className="mt-6 rounded-xl bg-navy p-5 text-white">
   282→                    <p className="text-xs uppercase tracking-wider text-gold-300">
   283→                      Customer sees this during checkout:
   284→                    </p>
   285→                    <div className="mt-3 space-y-1 text-sm">
   286→                      <div className="flex justify-between">
   287→                        <span className="text-navy-100/70">Bank</span>
   288→                        <span className="font-medium">{app.bankName || '—'}</span>
   289→                      </div>
   290→                      <div className="flex justify-between">
   291→                        <span className="text-navy-100/70">Account Name</span>
   292→                        <span className="font-medium">{app.accountName || '—'}</span>

   295→                        <span className="text-navy-100/70">Account Number</span>

   297→                          {app.accountNumber || '—'}
   298→                        </span>
   299→                      </div>
   300→                    </div>
   301→                  </div>
   302→
   303→                  {/* Card payments (Paystack) status — read-only, env-derived.
   304→                   * Makes it obvious to the admin why customers only see the
   305→                   * transfer option at checkout. */}
   306→                  <div className="mt-4 flex items-start gap-3 rounded-xl border border-navy-100 bg-linen-50 p-4">
   307→                    <CreditCard className="mt-0.5 h-4 w-4 shrink-0 text-navy-300" />
   308→                    <div className="text-sm">
   309→                      <p className="font-semibold text-navy">
   310→                        Card payments (Paystack){' '}
   311→                        <span
   312→                          className={
   313→                            app.paystackAvailable
   314→                              ? 'ml-1 rounded-full bg-emerald-100 px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wide text-emerald-700'
   315→                              : 'ml-1 rounded-full bg-navy-100 px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wide text-navy-300'
   316→                          }
   317→                        >
   318→                          {app.paystackAvailable ? 'Live' : 'Unavailable'}
   319→                        </span>
   320→                      </p>
   321→                      <p className="mt-1 text-xs leading-relaxed text-navy-300">
   322→                        {app.paystackAvailable
   323→                          ? 'Customers can pay by card — the option is selectable at checkout.'
   324→                          : 'Customers currently see the card option greyed out with “not available at the moment” and pay by bank transfer only. To enable cards, add PAYSTACK_SECRET_KEY in your deployment environment — it switches on automatically.'}
   325→                      </p>
   326→                    </div>
   327→                  </div>
   328→                </>
   329→              )}
   330→            </CardContent>
   331→          </Card>
   332→        </TabsContent>
   333→
   334→        {/* CONTACT TAB — server-backed, same as bank */}
   335→        <TabsContent value="contact" className="mt-4">
   336→          <Card className="border-navy-100 shadow-navy">
   337→            <CardHeader>
   338→              <CardTitle className="flex items-center gap-2 font-serif text-navy">
   339→                <Phone className="h-4 w-4 text-gold-400" /> Contact Information
   340→              </CardTitle>
   341→              <p className="text-xs text-navy-300">
   342→                Shown at checkout and on the offers strip — saved on the server for

   344→              </p>
   345→            </CardHeader>
   346→            <CardContent className="space-y-4">
   347→              {!app ? (
   348→                <p className="py-6 text-center text-sm text-navy-300">Loading contact details…</p>

   350→                <>
   351→                  <div>

   396→              </p>

   398→            <CardContent className="space-y-5">
   399→              {!app ? (

   401→              ) : (
   402→                <>
   403→                  <div>
   404→                    <Label htmlFor="alerts-email" className="text-xs uppercase tracking-wide text-navy-300">

   516→              {(['Shirts', 'Trousers', 'Suits', 'Traditional', "Women's Wear", 'Outerwear', 'Household', 'Extras', 'Shoes', 'Other'] as const).map(
   517→                (cat) => (
