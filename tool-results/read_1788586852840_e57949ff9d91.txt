    67→/** Strong-but-typeable password: guaranteed lower+upper+digit+symbol. */
    68→function generatePassword(): string {
    69→  const lower = 'abcdefghjkmnpqrstuvwxyz'
    70→  const upper = 'ABCDEFGHJKMNPQRSTUVWXYZ'
    71→  const digits = '23456789'
    72→  const symbols = '!#$%&*?'
    73→  const all = lower + upper + digits + symbols
    74→  const pick = (chars: string) => chars[Math.floor(Math.random() * chars.length)]
    75→  const rand = (n: number) =>
    76→    Array.from({ length: n }, () => pick(all))
    77→  // One of each class + 10 more random, then shuffle.
    78→  const core = [pick(lower), pick(upper), pick(digits), pick(symbols), ...rand(10)]
    79→  for (let i = core.length - 1; i > 0; i--) {
    80→    const j = Math.floor(Math.random() * (i + 1))
    81→    ;[core[i], core[j]] = [core[j], core[i]]
    82→  }
    83→  return core.join('')
    84→}

    86→function initials(name: string): string {
    87→  return name
    88→    .split(' ')
    89→    .map((p) => p[0])
    90→    .slice(0, 2)
    91→    .join('')
    92→    .toUpperCase()
    93→}

   416→                  { id: target.id, accessStatus: 'REVOKED' },
   417→                  {
   418→                    onSuccess: () =>
   419→                      toast({
   420→                        title: 'Access revoked',

   422→                      }),
   423→                  }
   424→                )
   425→                setRevokeTarget(null)
   426→              }}
   427→            >
   428→              Revoke access
   429→            </AlertDialogAction>
   430→          </AlertDialogFooter>
   431→        </AlertDialogContent>
   432→      </AlertDialog>
   433→    </div>

   591→            variant="outline"
   592→            onClick={() => {
   593→              reset()
   594→              onClose()
   595→            }}
   596→          >
   597→            Cancel
   598→          </Button>
   599→          <Button
   600→            disabled={!valid || pending}
   601→            onClick={() =>
   602→              onSubmit({
   603→                name: name.trim(),
   604→                email: email.trim().toLowerCase(),
   605→                phone: phone.trim(),
   606→                password,
   607→                note: note.trim() || undefined,
   608→              })
   609→            }
   610→            className={cn('bg-gold-gradient font-semibold text-[#0A192F] hover:opacity-90')}
   611→          >
   612→            {pending ? 'Creating account…' : 'Create account & send invite'}
   613→          </Button>
   614→        </DialogFooter>

   623→
   624→function ResetPasswordDialog({
   625→  member,
   626→  pending,
   627→  onClose,
   628→  onSubmit,
   629→}: {
   630→  member: ApiStaff | null
   631→  pending: boolean
   632→  onClose: () => void
   633→  onSubmit: (password: string) => void
   634→}) {
   635→  const [password, setPassword] = useState('')
   636→
   637→  return (
   638→    <Dialog
   639→      open={!!member}
   640→      onOpenChange={(o) => {
   641→        if (!o) {
   642→          setPassword('')
   643→          onClose()
   644→        }
   645→      }}
   646→    >
   647→      <DialogContent className="sm:max-w-md">
   648→        <DialogHeader>
   649→          <DialogTitle>Reset {member?.name}&apos;s password</DialogTitle>
   650→          <DialogDescription>
   651→            Sets a new password and emails it to {member?.email}. If their access is
   652→            paused or revoked, it is restored at the same time.
   653→          </DialogDescription>
   654→        </DialogHeader>
   655→        <div>
   656→          <Label htmlFor="reset-password" className="text-xs uppercase tracking-wide text-navy-300">
   657→            New password
   658→          </Label>
   659→          <div className="mt-1.5 flex gap-2">
   660→            <Input
   661→              id="reset-password"
   662→              value={password}
   663→              onChange={(e) => setPassword(e.target.value)}
   664→              placeholder="At least 10 characters"
   665→              className="font-mono"
   666→            />
   667→            <Button
   668→              type="button"
   669→              variant="outline"
   670→              className="shrink-0"
   671→              onClick={() => {
   672→                const pw = generatePassword()
   673→                setPassword(pw)
   674→                navigator.clipboard?.writeText(pw).catch(() => {})
   675→                toast({ title: 'Password generated and copied' })
   676→              }}
   677→            >
   678→              <Copy className="mr-1.5 h-3.5 w-3.5" /> Generate
   679→            </Button>
   680→          </div>
   681→        </div>
   682→        <DialogFooter>
   683→          <Button
   684→            variant="outline"
   685→            onClick={() => {
   686→              setPassword('')
   687→              onClose()
   688→            }}
   689→          >
   690→            Cancel
   691→          </Button>
   692→          <Button
   693→            disabled={password.length < 10 || pending}
   694→            onClick={() => onSubmit(password)}
   695→            className="bg-gold-gradient font-semibold text-[#0A192F] hover:opacity-90"
   696→          >
   697→            {pending ? 'Resetting…' : 'Reset & email them'}
   698→          </Button>
   699→        </DialogFooter>
   700→      </DialogContent>
   701→    </Dialog>
   702→  )
   703→}
