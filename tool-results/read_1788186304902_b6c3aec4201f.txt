   503→                    <span className="text-xs text-navy-300">
   504→                      {filledCount} of {defs.length} filled
   505→                    </span>
   506→                  )}
   507→                </div>
   508→
   509→                {/* inputs */}
   510→                <div className="mt-4 grid gap-3 sm:grid-cols-2">
   511→                  {defs.map((d) => (

   513→                      <span className="text-xs font-semibold text-navy">{d.label}</span>

   538→                        <Check className="h-4 w-4" /> Saved on this device

   545→                  </Button>

   548→                      onClick={doClear}
   549→                      variant="outline"

   558→              {/* status panel */}

   561→                <p className="mt-4 font-serif text-xl font-semibold text-white">
   562→                  {hasValues(saved)
   563→                    ? `Saved: ${CATEGORY_META[saved.profile].label.toLowerCase()}`
   564→                    : 'Nothing saved yet'}
   565→                </p>
   566→                <p className="mt-2 text-sm leading-relaxed text-white/70">

   575→                    <Check className="mt-0.5 h-4 w-4 shrink-0 text-gold-300" />
   576→                    Private to this browser — nothing is uploaded
   577→                  </li>
   578→                  <li className="flex gap-2">
   579→                    <Check className="mt-0.5 h-4 w-4 shrink-0 text-gold-300" />
   580→                    Edit and re-save any time your numbers change
   581→                  </li>
   582→                  <li className="flex gap-2">

   585→                  </li>

   587→              </div>
   588→            </div>
   589→          </CardContent>
   590→        </Card>
   591→
   592→        {/* ---------------- CTA ---------------- */}
   593→        <div className="mt-12 rounded-2xl bg-navy px-6 py-10 text-center sm:px-10">
   594→          <h2 className="font-serif text-2xl font-semibold tracking-tight text-white sm:text-3xl">

   596→          </h2>
   597→          <p className="mx-auto mt-3 max-w-xl text-sm leading-relaxed text-white/70">
   598→            Book a pickup, add an alteration, and describe what needs changing — your saved
   599→            measurements ride along with the order. The seamstress assesses every piece,
   600→            calls to confirm, and quotes before a single stitch is sewn.
   601→          </p>
   602→          <div className="mt-6 flex flex-wrap justify-center gap-3">
   603→            <Link href="/book">
